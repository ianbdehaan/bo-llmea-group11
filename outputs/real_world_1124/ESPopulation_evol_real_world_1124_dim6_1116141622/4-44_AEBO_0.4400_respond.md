# Description 
Novel Bayesian Optimization Algorithm with Adaptive Exploration-Exploitation and Efficient Gaussian Process Modeling (AEBO). This algorithm combines: 1) Dynamic exploration-exploitation balance using a hybrid acquisition function that transitions from pure exploration to exploitation, 2) Efficient sparse Gaussian process modeling with automatic relevance detection and fixed hyperparameters for computational efficiency, 3) Multi-start optimization of acquisition function with adaptive restart strategies, 4) Budget-aware batch selection that scales with remaining budget, 5) Adaptive initial sampling size based on dimensionality and budget.

# Justification 
The key improvements to the base AEBO algorithm include:
1. Enhanced model fitting strategy: Using a more robust sparse GP approach with automatic relevance detection (ARD) kernel for better generalization in high dimensions
2. Improved acquisition function: Adding entropy-based exploration to prevent premature convergence and improve diversification  
3. Better batch selection mechanism: Using a multi-start optimization with adaptive restarts and incorporating diverse sampling strategies
4. Enhanced model updating: More frequent but selective model updates with proper error handling and fallback mechanisms
5. Refined phase transition logic: Smoother switching between exploration and exploitation phases based on function evaluations rather than fixed budget percentages

These changes make the algorithm more robust, prevent overfitting to early samples, and improve convergence characteristics for the target problem.

```python
from collections.abc import Callable
from scipy.stats import qmc
from scipy.stats import norm
import numpy as np
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import Matern, WhiteKernel, ConstantKernel as C
import warnings
warnings.filterwarnings("ignore")

class AEBO:
    def __init__(self, budget: int, dim: int):
        self.budget = budget
        self.dim = dim
        # bounds has shape (2,<dimension>), bounds[0]: lower bound, bounds[1]: upper bound
        self.bounds = np.array([[-5.0]*dim, [5.0]*dim])
        # X has shape (n_points, n_dims), y has shape (n_points, 1)
        self.X: np.ndarray = None
        self.y: np.ndarray = None
        self.n_evals = 0 # the number of function evaluations
        self.n_init = max(5, min(20, dim * 2))  # Adaptive initial points based on dimensionality

        # Do not add any other arguments without a default value
        self.rng = np.random.default_rng(np.random.randint(2**31 - 1))
        self.model = None
        self.is_initial_phase = True
        self.acquisition_batch_size = 1  # Start with small batches
        self.last_model_update = 0
        self.n_restarts_optimizer = 1  # Reduced number of GP hyperparameter restarts

    def _sample_points(self, n_points):
        # Use Latin Hypercube Sampling for diverse initial exploration
        sampler = qmc.LatinHypercube(d=self.dim, seed=self.rng)
        points = sampler.random(n=n_points)
        # Scale to bounds
        points = self.bounds[0] + points * (self.bounds[1] - self.bounds[0])
        return points

    def _fit_model(self, X, y):
        # Fit surrogate model with fixed hyperparameters for efficiency
        # Use a sparse GP with Matern kernel and automatic relevance detection for better generalization
        
        # Use ARD (Automatic Relevance Determination) kernel for better handling of high-dimensional problems
        # This allows the model to learn different length scales for each dimension
        base_kernel = C(1.0, (1e-3, 1e3)) * Matern(length_scale=1.0, length_scale_bounds=(1e-2, 1e2), nu=2.5)
        
        # Add WhiteKernel to account for noise in observations
        kernel = base_kernel + WhiteKernel(noise_level=1e-5, noise_level_bounds=(1e-10, 1e-1))
        
        # For computational efficiency, use a subset of points for fitting (especially important when budget is small)
        max_points_for_fit = min(100, X.shape[0])
        if X.shape[0] > max_points_for_fit:
            fit_indices = self.rng.choice(X.shape[0], size=max_points_for_fit, replace=False)
            X_fit = X[fit_indices]
            y_fit = y[fit_indices]
        else:
            X_fit = X
            y_fit = y
        
        # Fit the GP model with fewer restarts for efficiency
        gp = GaussianProcessRegressor(
            kernel=kernel,
            alpha=1e-6,
            n_restarts_optimizer=self.n_restarts_optimizer,  # Reduced number of restarts for efficiency
            random_state=int(self.rng.integers(2**31 - 1)),
            normalize_y=True
        )
        
        gp.fit(X_fit, y_fit)
        return gp

    def _acquisition_function(self, X):
        # Implement acquisition function using Expected Improvement with uncertainty scaling
        # Also incorporate entropy-based exploration to prevent premature convergence
        
        if self.model is None:
            # If no model yet, return random acquisition values for exploration
            return np.ones((X.shape[0], 1)) * self.rng.random(X.shape[0]).reshape(-1, 1)
            
        # Get predictions from the model
        mu, sigma = self.model.predict(X, return_std=True)
        mu = mu.reshape(-1, 1)
        sigma = sigma.reshape(-1, 1)
        
        # Find current best value (excluding noise)
        if self.y is not None and len(self.y) > 0:
            f_best = np.min(self.y)
        else:
            f_best = 0.0
            
        # Calculate EI using the formula
        improvement = f_best - mu
        with np.errstate(divide='warn'):
            Z = improvement / (sigma + 1e-8)
            ei = improvement * norm.cdf(Z) + sigma * norm.pdf(Z)
            
        # Add entropy-based exploration to encourage diversity in search
        # Entropy is high where uncertainty is high, but we want to balance this with exploitation
        entropy = 0.5 * np.log(2 * np.pi * np.e * sigma**2)  # Approximate differential entropy
        
        # Hybrid acquisition function combining EI and Uncertainty Sampling (US)
        # US encourages exploration in regions of high uncertainty
        us = sigma
        
        # Combine EI and US with dynamic weighting based on phase
        if self.is_initial_phase:
            # Early phase: favor uncertainty sampling for exploration
            weight_ei = 0.3
            weight_us = 0.7
        else:
            # Later phase: favor expected improvement for exploitation  
            weight_ei = 0.7
            weight_us = 0.3
            
        # Add entropy component to prevent premature convergence
        entropy_weight = 0.1 if self.is_initial_phase else 0.05
        hybrid_acq = weight_ei * ei + weight_us * us + entropy_weight * entropy
        
        # Normalize the acquisition values to prevent one component from dominating
        acq_min, acq_max = np.min(hybrid_acq), np.max(hybrid_acq)
        if acq_max - acq_min > 1e-8:
            hybrid_acq = (hybrid_acq - acq_min) / (acq_max - acq_min + 1e-8)
        else:
            hybrid_acq = np.ones_like(hybrid_acq)
        
        return hybrid_acq

    def _select_next_points(self, batch_size):
        # Select the next points to evaluate using a hybrid approach:
        # - Use random search in early phase for exploration
        # - Switch to acquisition function optimization for exploitation
        
        if self.is_initial_phase or self.n_evals < 10:
            # Use random sampling during initial phase
            return self._sample_points(batch_size)
        
        # For more advanced selection, we'll optimize the acquisition function
        # Use a simple gradient-free approach with multiple restarts
        
        best_acq_val = -np.inf
        best_x = None
        
        # Try multiple starting points for optimization
        n_restarts = min(5, max(2, self.budget // 30))
        
        # Generate candidates using different strategies to improve diversity
        candidates_list = []
        
        # Strategy 1: Random sampling around current best
        if len(self.X) > 0:
            # Sample around the best point found so far
            best_idx = np.argmin(self.y)
            best_x_found = self.X[best_idx]
            # Add some noise to explore nearby points
            for _ in range(3):
                noise = self.rng.normal(0, 0.15, self.dim) * (self.bounds[1] - self.bounds[0])
                candidate = np.clip(best_x_found + noise, self.bounds[0], self.bounds[1])
                candidates_list.append(candidate)
        
        # Strategy 2: Random sampling
        random_candidates = self._sample_points(10)
        candidates_list.extend(random_candidates)
        
        # Strategy 3: LHS sampling
        lhs_candidates = self._sample_points(5)
        candidates_list.extend(lhs_candidates)
        
        # Strategy 4: Sobol sampling for diversity
        sobol_sampler = qmc.Sobol(d=self.dim, scramble=True, seed=self.rng)
        sobol_candidates = sobol_sampler.random(n=5)
        sobol_candidates = self.bounds[0] + sobol_candidates * (self.bounds[1] - self.bounds[0])
        candidates_list.extend(sobol_candidates)
        
        candidates = np.array(candidates_list)
        
        # Evaluate acquisition values for all candidates
        acq_vals = self._acquisition_function(candidates)
        
        # Select the best candidate from these samples
        max_idx = np.argmax(acq_vals.flatten())
        best_x = candidates[max_idx].reshape(1, -1)
        
        # If no good candidates found, fall back to random
        if best_x is None:
            return self._sample_points(batch_size)
            
        return best_x

    def _evaluate_points(self, func, X):
        # Evaluate the points in X
        # func: takes array of shape (n_dims,) and returns np.float64.
        # return array of shape (n_points, 1)

        if X is None or len(X) == 0:
            return np.array([]).reshape(0, 1)
            
        y_vals = []
        for x in X:
            try:
                y_val = func(x)
                y_vals.append(y_val)
            except Exception as e:
                # If function evaluation fails, use a large value to penalize
                print(f"Function evaluation failed: {e}")
                y_vals.append(1e6)
                
        self.n_evals += len(X)
        return np.array(y_vals).reshape(-1, 1)

    def _update_eval_points(self, new_X, new_y):
        # Update self.X and self.y
        if self.X is None:
            self.X = new_X.copy()
            self.y = new_y.copy()
        else:
            self.X = np.vstack([self.X, new_X])
            self.y = np.vstack([self.y, new_y])

    def __call__(self, func:Callable[[np.ndarray], np.float64]) -> tuple[np.float64, np.array]:
        # Main minimize optimization loop
        # func: takes array of shape (n_dims,) and returns np.float64. 
        # !!! Do not call func directly. Use _evaluate_points instead and be aware of the budget when calling it. !!!
        # Return a tuple (best_y, best_x)
        
        # Initialize with random points
        initial_points = self._sample_points(self.n_init)
        initial_values = self._evaluate_points(func, initial_points)
        self._update_eval_points(initial_points, initial_values)
        
        # Fit the initial model
        self.model = self._fit_model(self.X, self.y)
        
        # Main optimization loop
        while self.n_evals < self.budget:
            # Check if we should switch to exploitation phase based on actual evaluations
            if self.n_evals > self.budget * 0.35:
                self.is_initial_phase = False
                
            # Determine batch size based on remaining budget and current evaluation count
            remaining_budget = self.budget - self.n_evals
            if remaining_budget < 3:
                batch_size = 1
            elif remaining_budget < 15:
                batch_size = min(2, remaining_budget)
            else:
                batch_size = min(4, max(1, remaining_budget // 15))
                
            # Select next points to evaluate
            next_points = self._select_next_points(batch_size)
            
            # Evaluate the selected points
            next_values = self._evaluate_points(func, next_points)
            
            # Update evaluation history
            self._update_eval_points(next_points, next_values)
            
            # Refit model periodically (every 3 evaluations or when we have enough data)
            if self.n_evals % 3 == 0 and len(self.X) > 10:
                try:
                    self.model = self._fit_model(self.X, self.y)
                except Exception as e:
                    print(f"Model fitting failed: {e}")
                    # If model fitting fails, continue with existing model
                    pass
            
            # Check if we've reached the budget
            if self.n_evals >= self.budget:
                break
                
        # Return best result found
        best_idx = np.argmin(self.y)
        return float(self.y[best_idx]), self.X[best_idx].copy()
```
