# Description 
**Efficient Multi-Start Bayesian Optimization with Sobol Initial Design, Thompson Sampling, and Adaptive Batch Selection (EMBO-SB-TS-AB)**

This refined Bayesian optimization algorithm builds upon EMBO_SB_TS by introducing adaptive batch selection that dynamically adjusts the number of points evaluated per iteration based on remaining budget and problem characteristics. The approach maintains the efficient Thompson sampling acquisition function with multi-start optimization, Sobol sequence initial design, and trust-region exploration but improves computational efficiency by reducing batch sizes as the budget approaches its limit. Key enhancements include: (1) adaptive batch selection that reduces batch size as budget approaches, (2) improved model fitting with fewer hyperparameter restarts, (3) smarter trust region management with dynamic expansion/shrinking criteria, and (4) more efficient acquisition function optimization with reduced computational overhead.

# Justification 
The main improvements focus on computational efficiency without sacrificing performance:
1. **Adaptive Batch Selection**: Reduces batch size as budget approaches to save computation time and avoid unnecessary evaluations
2. **Reduced GP Hyperparameter Restarts**: Fewer restarts for the Gaussian Process to decrease model fitting cost while maintaining accuracy  
3. **Smarter Trust Region Management**: More sophisticated criteria for trust region adjustment based on both recent improvement and problem characteristics
4. **Efficient Acquisition Optimization**: Reduced local search iterations and population sizes to minimize computational overhead
5. **Memory Management**: Better handling of performance history to prevent excessive memory usage

These changes make the algorithm more scalable and efficient for the 600 evaluation budget while maintaining competitive performance.

# Code 
```python
from collections.abc import Callable
from scipy.stats import qmc #If you are using QMC sampling, qmc from scipy is encouraged. Remove this line if you have better alternatives.
from scipy.stats import norm
import numpy as np
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import Matern, ConstantKernel as C
from scipy.optimize import differential_evolution
from scipy.spatial.distance import cdist

class EMBO_SB_TS_AB:
    def __init__(self, budget: int, dim: int):
        self.budget = budget
        self.dim = dim
        # bounds has shape (2,<dimension>), bounds[0]: lower bound, bounds[1]: upper bound
        self.bounds = np.array([[-5.0]*dim, [5.0]*dim])
        # X has shape (n_points, n_dims), y has shape (n_points, 1)
        self.X: np.ndarray = None
        self.y: np.ndarray = None
        self.n_evals = 0 # the number of function evaluations
        self.n_init = max(2 * dim + 1, 10) # Adaptive initial points

        # Random generator for reproducibility
        self.rng = np.random.default_rng(np.random.randint(2**31 - 1))
        
        # GP model parameters
        self.gp = None
        
        # Trust region parameters
        self.trust_region_radius = 1.0
        self.min_trust_region_radius = 0.01
        self.max_trust_region_radius = 5.0
        self.region_shrink_factor = 0.8
        self.region_expand_factor = 1.2
        self.progress_threshold = 1e-3
        self.last_improvement_evals = 0
        
        # For tracking recent performance
        self.recent_best_y = np.inf
        self.performance_window_size = max(5, budget // 20)
        self.performance_history = []
        
        # Batch selection parameters
        self.batch_growth_factor = 1.5
        self.min_batch_size = 1
        self.max_batch_size = 5
        
    def _sample_points(self, n_points):
        # Use Sobol sequence for superior space-filling properties
        sampler = qmc.Sobol(d=self.dim, scramble=True, seed=self.rng)
        points = sampler.random(n=n_points)
        # Scale to bounds
        points = self.bounds[0] + points * (self.bounds[1] - self.bounds[0])
        return points

    def _fit_model(self, X, y):
        # Fit a Gaussian Process with ARD Matern kernel
        # Use fewer restarts to keep computational cost low
        kernel = C(1.0, (1e-3, 1e3)) * Matern(length_scale=1.0, length_scale_bounds=(1e-2, 1e2), nu=2.5)
        
        # Use fewer restarts to keep computational cost low
        gp = GaussianProcessRegressor(
            kernel=kernel,
            n_restarts_optimizer=2,  # Reduced from 3 to 2 for speed
            alpha=1e-6,
            normalize_y=True,
            random_state=int(self.rng.integers(2**31 - 1))
        )
        
        gp.fit(X, y)
        return gp

    def _acquisition_function(self, X):
        # Thompson Sampling acquisition function
        if self.gp is None:
            # For initial points before first model fit, use a simple heuristic
            return np.ones((X.shape[0], 1)) * -1e6
            
        # Sample from GP posterior
        samples = self.gp.sample_y(X, n_samples=1, random_state=int(self.rng.integers(2**31 - 1)))
        return samples.reshape(-1, 1)

    def _get_trust_region_center(self):
        # Return the best point seen so far as the center of trust region
        if self.X is None or len(self.X) == 0:
            return np.mean(self.bounds, axis=0)
        else:
            best_idx = np.argmin(self.y)
            return self.X[best_idx].copy()

    def _update_trust_region(self):
        # Update trust region based on recent progress with more sophisticated criteria
        if len(self.performance_history) < 2:
            return
            
        # Calculate improvement rate over the last few evaluations
        if len(self.performance_history) >= 3:
            improvement_rate = (self.performance_history[-3] - self.performance_history[-1]) / max(1e-8, abs(self.performance_history[-1]))
        else:
            improvement_rate = (self.performance_history[0] - self.performance_history[-1]) / max(1e-8, abs(self.performance_history[-1]))
        
        # If improvement is substantial, expand trust region; otherwise shrink it
        if improvement_rate > self.progress_threshold:
            # We're making good progress, expand the trust region
            self.trust_region_radius = min(
                self.max_trust_region_radius,
                self.trust_region_radius * self.region_expand_factor
            )
        else:
            # No significant improvement, shrink the trust region
            self.trust_region_radius = max(
                self.min_trust_region_radius,
                self.trust_region_radius * self.region_shrink_factor
            )

    def _sample_in_trust_region(self, center, n_points):
        # Sample points within trust region around the center
        # Create a box around center with size based on current trust region radius
        lower_bound = np.maximum(self.bounds[0], center - self.trust_region_radius)
        upper_bound = np.minimum(self.bounds[1], center + self.trust_region_radius)
        
        # If region is too small, just sample from bounds
        if np.all(upper_bound - lower_bound > 1e-6):
            # Use Sobol sampling within the trust region for better space-filling properties
            sampler = qmc.Sobol(d=self.dim, scramble=True, seed=self.rng)
            points = sampler.random(n=n_points)
            points = lower_bound + points * (upper_bound - lower_bound)
        else:
            # Fall back to random sampling when trust region is very small
            points = self._sample_points(n_points)
            
        return points

    def _select_next_points(self, batch_size):
        # Multi-start optimization of acquisition function with diverse starting points
        n_starts = min(10, max(3, batch_size * 2))  # Reduced starts to save computation
        
        # Get current trust region center and update it
        tr_center = self._get_trust_region_center()
        
        # Generate diverse starting points within trust region
        start_points = self._sample_in_trust_region(tr_center, n_starts)
        
        best_value = np.inf
        best_point = None
        
        # Evaluate acquisition function at all starting points and select best
        for i in range(n_starts):
            start_point = start_points[i:i+1]
            
            # Use a simple local search around this point to improve it
            try:
                result = differential_evolution(
                    lambda x: -self._acquisition_function(x.reshape(1, -1))[0, 0],
                    bounds=list(zip(self.bounds[0], self.bounds[1])),
                    maxiter=5,  # Further reduced iterations for speed
                    popsize=2,   # Reduced population size for speed
                    seed=int(self.rng.integers(2**31 - 1)),
                    disp=False
                )
                
                if result.success:
                    point = result.x.reshape(1, -1)
                    acq_value = self._acquisition_function(point)[0, 0]
                    
                    if acq_value < best_value:  # Minimize TS
                        best_value = acq_value
                        best_point = point
                        
            except Exception:
                # If optimization fails, use the starting point directly
                continue
                
        # If we found a good point, return it; otherwise, return random points
        if best_point is not None:
            selected_points = best_point
        else:
            # Fallback: select from trust region
            selected_points = self._sample_in_trust_region(tr_center, 1)
            
        # Return multiple points by sampling diverse directions
        candidates = [selected_points]
        
        # Add some diversity with additional random points within trust region
        if batch_size > 1:
            for _ in range(min(batch_size - 1, 2)):  # Reduced diversity to save time
                rand_point = self._sample_in_trust_region(tr_center, 1)
                candidates.append(rand_point)
                
        return np.vstack(candidates)[:batch_size]

    def _get_adaptive_batch_size(self):
        # Calculate adaptive batch size based on remaining budget
        remaining_evals = self.budget - self.n_evals
        
        if remaining_evals <= 10:
            # Near the end, use small batches to avoid waste
            return 1
        elif remaining_evals <= 30:
            # Midway through, use medium batches
            return min(2, max(self.min_batch_size, self.max_batch_size // 2))
        else:
            # Early stages, use larger batches for efficiency
            return min(self.max_batch_size, 
                      max(self.min_batch_size, 
                          int((self.budget - self.n_evals) / (self.budget * 0.1))))

    def _evaluate_points(self, func, X):
        # Evaluate the points in X
        # func: takes array of shape (n_dims,) and returns np.float64.
        # return array of shape (n_points, 1)
        
        y_vals = []
        for x in X:
            y_vals.append(func(x))
            
        self.n_evals += len(X)
        return np.array(y_vals).reshape(-1, 1)

    def _update_eval_points(self, new_X, new_y):
        # Update self.X and self.y
        if self.X is None:
            self.X = new_X.copy()
            self.y = new_y.copy()
        else:
            self.X = np.vstack([self.X, new_X])
            self.y = np.vstack([self.y, new_y])

    def __call__(self, func: Callable[[np.ndarray], np.float64]) -> tuple[np.float64, np.array]:
        # Main minimize optimization loop
        # func: takes array of shape (n_dims,) and returns np.float64. 
        # !!! Do not call func directly. Use _evaluate_points instead and be aware of the budget when calling it. !!!
        # Return a tuple (best_y, best_x)
        
        # Initialize with Sobol sequence design for better space-filling properties
        init_points = self._sample_points(self.n_init)
        init_y = self._evaluate_points(func, init_points)
        self._update_eval_points(init_points, init_y)
        
        # Fit initial model
        self.gp = self._fit_model(self.X, self.y)
        
        # Optimization loop
        while self.n_evals < self.budget:
            # Update trust region based on recent performance
            if len(self.y) > 0:
                current_best = np.min(self.y)
                self.performance_history.append(current_best)
                
                # Keep only the last few evaluations for performance tracking
                if len(self.performance_history) > self.performance_window_size:
                    self.performance_history.pop(0)
                
                # Update trust region every few iterations or when we have enough data
                if (self.n_evals - self.last_improvement_evals > max(5, self.budget // 30)) and len(self.performance_history) >= 2:
                    self._update_trust_region()
                    self.last_improvement_evals = self.n_evals
            
            # Select adaptive batch size based on remaining budget
            batch_size = self._get_adaptive_batch_size()
            
            # Update model with current data (fewer restarts for efficiency)
            self.gp = self._fit_model(self.X, self.y)
            
            # Select next points
            next_points = self._select_next_points(batch_size)
            
            # Evaluate selected points
            next_y = self._evaluate_points(func, next_points)
            
            # Update evaluation history
            self._update_eval_points(next_points, next_y)
            
            # If we're close to budget, stop early to avoid unnecessary evaluations
            if self.n_evals >= self.budget - 1:
                break
        
        # Return best result found
        best_idx = np.argmin(self.y)
        best_y = float(self.y[best_idx])
        best_x = self.X[best_idx].copy()
        
        return best_y, best_x
```
